CÓDIGO (sensor_litosfera.ino) 
#include <DHT.h>

#define SENSOR_PIN A0     // Pin analógico lectura humedad suelo (Litosfera)
#define DHT_PIN 2        // Pin digital sensor humedad/temp aire (Biosfera)
#define DHT_TYPE DHT11

DHT dht(DHT_PIN, DHT_TYPE);

void setup() {
  Serial.begin(9600);
  dht.begin();
  pinMode(SENSOR_PIN, INPUT);
  Serial.println("--- Sistema de Monitoreo Biosfera-Litosfera Inicializado ---");
}

void loop() {
  int humedadSueloRaw = analogRead(SENSOR_PIN);
  float humedadSueloPorcentaje = map(humedadSueloRaw, 1023, 0, 0, 100);
  
  float tempAire = dht.readTemperature();
  float humAire = dht.readHumidity();

  Serial.print("Humedad Suelo (Litosfera): ");
  Serial.print(humedadSueloPorcentaje);
  Serial.println("%");

  Serial.print("Temperatura Aire (Biosfera): ");
  Serial.print(tempAire);
  Serial.println(" °C");

  delay(2000);
}
