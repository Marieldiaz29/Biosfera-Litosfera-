CÓDIGO (sensor_litosfera.ino) 
#define SENSOR_LITOSFERA_PIN A0  // Sensor de humedad de suelo
#define LED_INDICADOR_PIN 8      // LED indicador de nivel de humedad

void setup() {
  Serial.begin(9600);
  pinMode(SENSOR_LITOSFERA_PIN, INPUT);
  pinMode(LED_INDICADOR_PIN, OUTPUT);
  Serial.println("--- Litosfera & Biosfera: Sistema Listo ---");
}

void loop() {
  int lecturAnalogica = analogRead(SENSOR_LITOSFERA_PIN);
  
  if (lecturAnalogica > 500) {
    digitalWrite(LED_INDICADOR_PIN, HIGH); // Activa LED encendido
  } else {
    digitalWrite(LED_INDICADOR_PIN, LOW);
  }

  Serial.print("Valor Humedad Litosfera: ");
  Serial.println(lecturAnalogica);
  delay(1000);
}
